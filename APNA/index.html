Google Sheets Live Sync Hub
Enter your deployed Google Apps Script Web App URL to receive live bookings & expenses.
Copy-Paste Google Apps Script Code:

In Google Sheets > Extensions > Apps Script, paste this script and click Deploy as Web App:

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.id || "EXPENSE",
    data.createdAt || new Date().toISOString(),
    data.custName || data.category || "",
    data.custPhone || "",
    data.serviceType || "",
    data.pickupLocation || "",
    data.dropLocation || "",
    data.vehicleType || "",
    data.driverAssigned || "",
    data.totalFare || data.amount || 0,
    data.advancePaid || 0,
    data.balanceDue || 0,
    data.paymentMode || "",
    data.bookingStatus || "Recorded"
  ]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
